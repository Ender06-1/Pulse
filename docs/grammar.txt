program =
  | stmt* EOF
  ;

stmt =
  | "print" expr ";"
  | "var" ident "=" expr ";"
  | "if" expr stmt_block ( "else" stmt_block )?
  | "for" stmt_block
  ;

stmt_block =
  | "{" stmt+ "}"
  ;

expr =
  | add_expr
  ;

add_expr =
  | mul_expr ( ( "+" | "-" ) mul_expr )*
  ;

mul_expr =
  | prim_expr ( ( "*" | "/" | "%" ) prim_expr )*
  ;

prim_expr =
  | integer
  | ident
  ;

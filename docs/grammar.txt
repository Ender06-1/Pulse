program =
  | fun_def+ EOF
  ;

fun_def =
  | "fn" ident "(" ( ident ( "," ident )* )? ")" ":" ident stmt_block
  ;

stmt =
  | "print" expr ";"
  | "var" ident "=" expr ";"
  | "if" expr stmt_block ( "else" stmt_block )?
  | "for" expr stmt_block
  | ident "=" expr ";"
  | "break" ";"
  | "continue" ";"
  | "return" expr? ";"
  | expr ";"
  ;

stmt_block =
  | "{" stmt+ "}"
  ;

expr =
  | equal_expr
  ;

equal_expr =
  | rela_expr ( ( "==" | "!=" ) rela_expr )*
  ;

rela_expr =
  | add_expr ( ( ">" | "<" | ">=" | "<=" add_expr )*
  ;

add_expr =
  | mul_expr ( ( "+" | "-" ) mul_expr )*
  ;

mul_expr =
  | prim_expr ( ( "*" | "/" | "%" ) prim_expr )*
  ;

prim_expr =
  | integer
  | ident ( "(" ( expr ( "," expr )* )? ")" )?
  | "(" expr ")"
  ;

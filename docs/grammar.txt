program =
  | fun_def+ EOF
  ;

fun_def =
  | "fn" ident "(" ( arg ( "," arg )* )? ")" ":" type_ref stmt_block
  ;

arg =
  | ident ":" type_ref
  ;

type_ref =
  | "*"? ident
  ;

stmt =
  | "print" expr ";"
  | "var" ident "=" expr ";"
  | "if" expr stmt_block ( "else" stmt_block )?
  | "for" expr stmt_block
  | lvalue "=" expr ";"
  | "break" ";"
  | "continue" ";"
  | "return" expr? ";"
  | expr ";"
  ;

stmt_block =
  | "{" stmt+ "}"
  ;

lvalue =
  | ident
  | "*" ident
  ;

expr =
  | equal_expr
  ;

equal_expr =
  | rela_expr ( ( "==" | "!=" ) rela_expr )*
  ;

rela_expr =
  | add_expr ( ( ">" | "<" | ">=" | "<=" add_expr )*
  ;

add_expr =
  | mul_expr ( ( "+" | "-" ) mul_expr )*
  ;

mul_expr =
  | unary_expr ( ( "*" | "/" | "%" ) unary_expr )*
  ;

unary_expr =
  | ( "*" | "&" )? prim_expr
  ;

prim_expr =
  | integer_lit
  | "null"
  | ident ( "(" ( expr ( "," expr )* )? ")" )?
  | "(" expr ")"
  ;
